\begin{frame}{miniKanren vs. semiringKanren}

	\begin{columns}
		\column{0.5\textwidth}
		miniKanren
		\begin{itemize}
			\item evaluation via stream search (top-down)
			\item dyamically typed (s-expressions)
			\item relations are unweighted (usually)
		\end{itemize}

		\column{0.5\textwidth}
		semiringKanren
		\begin{itemize}
			\item evaluation via multidimensional array operations (bottom-up)
			\item statically typed (finite algebraic data types)
			\item relations are weighted
		\end{itemize}

	\end{columns}

\end{frame}

\begin{frame}
	
	semiringKanren represents relations as finite multidimensional arrays.\\

	\vspace{1ex}

	Polymorphic relations need to represent arbitrarily many specifically-typed relations.\\

	\vspace{1ex}

		\item<1-> Replace polymorphic relation \tikzmarknode{monomorphic-relation-calls}{calls} with concrete (monomorphic) relation calls

		\item<3-> Expand large-enough monomorphic relation calls back to the right types
	\end{enumerate}
